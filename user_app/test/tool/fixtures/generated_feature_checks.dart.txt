class MemoryCache extends PreferenceCache {
  String? value;
  bool failRead = false;
  bool failWrite = false;
  int writes = 0;
  @override Future<String?> get(String key, {bool secure = false}) async {
    if (failRead) throw StateError('read failed');
    return value;
  }
  @override Future<void> put(String key, String value, Duration duration, {bool secure = false}) async {
    writes++;
    if (failWrite) throw StateError('write failed');
    this.value = value;
  }
}
class RemoteSource implements SampleItemsRemoteDataSource {
  int calls = 0;
  Object? error;
  List<SampleItemsItemData> items = const [SampleItemsItemData(id: '1', name: 'Fresh item')];
  @override Future<List<SampleItemsItemData>> fetchItems() async {
    calls++;
    final failure = error;
    if (failure != null) throw failure;
    return items;
  }
}
class Remote implements SampleItemsRepository {
  int calls = 0;
  bool? forced;
  bool shouldThrow = false;
  Completer<Either<Failure, SampleItemsItemList>>? pending;
  Either<Failure, SampleItemsItemList> result = const Right(SampleItemsItemList(items: [SampleItemsItem(id: '1', name: 'Fresh item')]));
  @override ResultFuture<SampleItemsItemList> getSampleItemsList({bool forceRefresh = false}) async {
    calls++;
    forced = forceRefresh;
    if (shouldThrow) throw StateError('unexpected error');
    if (pending != null) return pending!.future;
    return result;
  }
}
SampleItemsRepositoryImpl repository(MemoryCache cache, RemoteSource remote) =>
    SampleItemsRepositoryImpl(remote, SampleItemsLocalDataSource(cache));

void main() {
  tearDown(() => Get.reset());

  test('cache hit skips remote; refresh bypasses and replaces cache', () async {
    final cache = MemoryCache()..value = '[{"id":"0","name":"Cached item"}]';
    final remote = RemoteSource();
    final useCase = SampleItemsUseCase(repository(cache, remote));
    final first = await useCase(false);
    expect(first.getOrElse(() => const SampleItemsItemList()).items.single.name, 'Cached item');
    expect(remote.calls, 0);
    final refreshed = await useCase(true);
    expect(remote.calls, 1);
    expect(refreshed.getOrElse(() => const SampleItemsItemList()).items.single.name, 'Fresh item');
    expect((jsonDecode(cache.value!) as List).single['name'], 'Fresh item');
  });

  test('corrupt cache and storage failures do not hide remote success', () async {
    for (final malformed in ['invalid json', '{}', '[{"id":123}]']) {
      final cache = MemoryCache()..value = malformed;
      final remote = RemoteSource();
      expect((await repository(cache, remote).getSampleItemsList()).isRight(), isTrue);
      expect(remote.calls, 1);
    }
    final cache = MemoryCache()..failRead = true..failWrite = true;
    expect((await repository(cache, RemoteSource()).getSampleItemsList()).isRight(), isTrue);
  });

  test('failed refresh preserves cache and returns the failure', () async {
    final cache = MemoryCache()..value = '[]';
    final remote = RemoteSource()..error = const ServerException('Offline');
    final result = await repository(cache, remote).getSampleItemsList(forceRefresh: true);
    expect(result.isLeft(), isTrue);
    result.fold((failure) => expect(failure.message, 'Offline'), (_) => fail('Expected failure'));
    expect(cache.value, '[]');
    expect(cache.writes, 0);
  });

  test('remote parsing and connection errors map to typed failures', () async {
    final remote = RemoteSource()..error = const FormatException('bad');
    final parsing = await repository(MemoryCache(), remote).getSampleItemsList();
    parsing.fold((failure) => expect(failure, isA<ParsingFailure>()), (_) => fail('Expected failure'));
    remote.error = StateError('offline');
    final connection = await repository(MemoryCache(), remote).getSampleItemsList();
    connection.fold((failure) => expect(failure, isA<ConnectionFailure>()), (_) => fail('Expected failure'));
  });

  test('controller clears loading after thrown failures and can retry', () async {
    final remote = Remote()..shouldThrow = true;
    Get.put(SampleItemsUseCase(remote));
    final controller = SampleItemsScreenController();
    await controller.getData();
    expect(controller.isLoading.value, isFalse);
    expect(controller.errorMessage.value, isNotNull);
    remote.shouldThrow = false;
    await controller.refreshData();
    expect(remote.forced, isTrue);
    expect(controller.errorMessage.value, isNull);
    expect(controller.items.single.name, 'Fresh item');
  });

  testWidgets('loading, error, retry, data, refresh error and empty states', (tester) async {
    final remote = Remote()..pending = Completer<Either<Failure, SampleItemsItemList>>();
    Get.put(SampleItemsUseCase(remote));
    final controller = Get.put(SampleItemsScreenController());
    await tester.pumpWidget(const GetMaterialApp(home: SampleItemsScreen()));
    expect(find.byType(CircularProgressIndicator), findsOneWidget);
    await controller.getData();
    expect(remote.calls, 1);
    remote.pending!.complete(const Left(ServerFailure('Offline')));
    await tester.pumpAndSettle();
    expect(find.text('Offline'), findsOneWidget);
    expect(find.text('Retry'), findsOneWidget);
    remote.pending = null;
    await tester.tap(find.text('Retry'));
    await tester.pumpAndSettle();
    expect(remote.forced, isTrue);
    expect(find.text('Fresh item'), findsOneWidget);
    expect(find.text('Offline'), findsNothing);
    final beforeRefresh = remote.calls;
    await tester.drag(find.byType(CustomScrollView), const Offset(0, 400));
    await tester.pumpAndSettle();
    expect(remote.calls, beforeRefresh + 1);
    remote.result = const Left(ServerFailure('Refresh failed'));
    await controller.refreshData();
    await tester.pumpAndSettle();
    expect(find.text('Fresh item'), findsOneWidget);
    expect(find.text('Refresh failed'), findsOneWidget);
    remote.result = const Right(SampleItemsItemList());
    await tester.tap(find.text('Retry'));
    await tester.pumpAndSettle();
    expect(find.text('No data found'), findsOneWidget);
    expect(find.text('Refresh failed'), findsNothing);
    await tester.pumpWidget(const SizedBox.shrink());
  });
}
